-- JAS: shared boards (run once in Supabase → SQL Editor → New query → Run)

create table if not exists public.profiles (
  id uuid primary key references auth.users on delete cascade,
  email text unique not null,
  name text not null default ''
);

create table if not exists public.boards (
  id uuid primary key default gen_random_uuid(),
  owner uuid not null default auth.uid() references auth.users on delete cascade,
  title text not null default 'Untitled board',
  data jsonb not null default '{}'::jsonb,
  updated_at timestamptz not null default now(),
  updated_by uuid default auth.uid()
);

create table if not exists public.board_members (
  board_id uuid references public.boards on delete cascade,
  user_id uuid references auth.users on delete cascade,
  role text not null default 'editor',
  added_at timestamptz not null default now(),
  primary key (board_id, user_id)
);

-- A profile row for every new account, so people can be invited by email
create or replace function public.handle_new_user() returns trigger
language plpgsql security definer set search_path = public as $$
begin
  insert into public.profiles (id, email, name)
  values (new.id, lower(new.email), coalesce(new.raw_user_meta_data->>'name', ''))
  on conflict (id) do nothing;
  return new;
end $$;
drop trigger if exists on_auth_user_created on auth.users;
create trigger on_auth_user_created after insert on auth.users
  for each row execute function public.handle_new_user();

-- The board owner is always a member
create or replace function public.add_owner_member() returns trigger
language plpgsql security definer set search_path = public as $$
begin
  insert into public.board_members (board_id, user_id, role) values (new.id, new.owner, 'owner')
  on conflict do nothing;
  return new;
end $$;
drop trigger if exists on_board_created on public.boards;
create trigger on_board_created after insert on public.boards
  for each row execute function public.add_owner_member();

create or replace function public.is_member(b uuid) returns boolean
language sql security definer stable set search_path = public as $$
  select exists (select 1 from board_members where board_id = b and user_id = auth.uid());
$$;

alter table public.profiles enable row level security;
alter table public.boards enable row level security;
alter table public.board_members enable row level security;

drop policy if exists "see own and co-member profiles" on public.profiles;
create policy "see own and co-member profiles" on public.profiles for select using (
  id = auth.uid() or exists (
    select 1 from public.board_members m1 join public.board_members m2 on m1.board_id = m2.board_id
    where m1.user_id = auth.uid() and m2.user_id = profiles.id));
drop policy if exists "update own profile" on public.profiles;
create policy "update own profile" on public.profiles for update using (id = auth.uid());

drop policy if exists "members read boards" on public.boards;
create policy "members read boards" on public.boards for select using (owner = auth.uid() or public.is_member(id));
drop policy if exists "create own boards" on public.boards;
create policy "create own boards" on public.boards for insert with check (owner = auth.uid());
drop policy if exists "members edit boards" on public.boards;
create policy "members edit boards" on public.boards for update using (public.is_member(id));
drop policy if exists "owner deletes boards" on public.boards;
create policy "owner deletes boards" on public.boards for delete using (owner = auth.uid());

drop policy if exists "members see members" on public.board_members;
create policy "members see members" on public.board_members for select using (public.is_member(board_id));
drop policy if exists "owner removes members or member leaves" on public.board_members;
create policy "owner removes members or member leaves" on public.board_members for delete using (
  user_id = auth.uid() or exists (select 1 from public.boards where id = board_id and owner = auth.uid()));

-- Invite someone by the email they signed up with
create or replace function public.add_board_member(b uuid, member_email text) returns json
language plpgsql security definer set search_path = public as $$
declare uid uuid;
begin
  if not public.is_member(b) then raise exception 'Not allowed'; end if;
  select id into uid from public.profiles where email = lower(trim(member_email));
  if uid is null then return json_build_object('ok', false, 'reason', 'no_account'); end if;
  insert into public.board_members (board_id, user_id) values (b, uid) on conflict do nothing;
  return json_build_object('ok', true, 'user_id', uid);
end $$;
